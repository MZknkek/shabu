// เพิ่มใน server.js หลัง GET /menus
// ======================================================

// POST - เพิ่มเมนู
app.post('/menus', (req, res) => {
    const {
        menu_name,
        description,
        category,
        price,
        image,
        status
    } = req.body;

    const sql = `
        INSERT INTO menu
        (menu_name, description, category, price, image, status)
        VALUES (?, ?, ?, ?, ?, ?)
    `;

    db.query(
        sql,
        [menu_name, description, category, price, image, status],
        (err, result) => {
            if (err) {
                return res.status(500).json({
                    success: false,
                    message: 'เพิ่มเมนูไม่สำเร็จ',
                    error: err.message
                });
            }

            res.json({
                success: true,
                message: 'เพิ่มเมนูสำเร็จ',
                menu_id: result.insertId
            });
        }
    );
});

// PUT - แก้ไขเมนู
app.put('/menus/:id', (req, res) => {
    const { id } = req.params;

    const {
        menu_name,
        description,
        category,
        price,
        image,
        status
    } = req.body;

    const sql = `
        UPDATE menu
        SET
            menu_name = ?,
            description = ?,
            category = ?,
            price = ?,
            image = ?,
            status = ?
        WHERE menu_id = ?
    `;

    db.query(
        sql,
        [
            menu_name,
            description,
            category,
            price,
            image,
            status,
            id
        ],
        (err, result) => {
            if (err) {
                return res.status(500).json({
                    success: false,
                    message: 'แก้ไขเมนูไม่สำเร็จ',
                    error: err.message
                });
            }

            if (result.affectedRows === 0) {
                return res.status(404).json({
                    success: false,
                    message: 'ไม่พบเมนูที่ต้องการแก้ไข'
                });
            }

            res.json({
                success: true,
                message: 'แก้ไขเมนูสำเร็จ'
            });
        }
    );
});

// DELETE - ลบเมนู
app.delete('/menus/:id', (req, res) => {
    const { id } = req.params;

    const sql = 'DELETE FROM menu WHERE menu_id = ?';

    db.query(sql, [id], (err, result) => {
        if (err) {
            return res.status(500).json({
                success: false,
                message: 'ลบเมนูไม่สำเร็จ',
                error: err.message
            });
        }

        if (result.affectedRows === 0) {
            return res.status(404).json({
                success: false,
                message: 'ไม่พบเมนูที่ต้องการลบ'
            });
        }

        res.json({
            success: true,
            message: 'ลบเมนูสำเร็จ'
        });
    });
});
